\begin{tcolorbox}[enhanced,colback=Fond,colframe=Fond,arc=0pt,boxrule=0pt,left=10pt,right=10pt,top=8pt,bottom=8pt]
\small Sujet d'entraînement rédigé par Le Collé (ce n'est pas une annale). Trois exercices indépendants, dans l'esprit d'un DS :
séries numériques, séries entières, et une équation différentielle résolue par les séries entières. Calculatrice interdite.
La qualité de la rédaction et la précision des justifications seront prises en compte.
\end{tcolorbox}

\Partie{Exercice 1 — Séries numériques}

\begin{enumerate}
\item Déterminer la nature des séries de terme général :
\begin{gather*}\text{(a)}\ u_n=1-\cos\frac1{\sqrt n}\qquad\text{(b)}\ u_n=\frac{(n!)^2\,3^n}{(2n)!}\\
\text{(c)}\ u_n=\frac{n^{\ln n}}{(\ln n)^n}\ (n\geq2)\qquad\text{(d)}\ u_n=(-1)^n\ln\left(1+\frac1n\right).\end{gather*}
\item Soit $v_n=\dfrac{(-1)^n}{\sqrt n+(-1)^n}$ pour $n\geq2$.
\begin{enumerate}[label=(\alph*)]
\item Le critère des séries alternées s'applique-t-il ?
\item Montrer que $v_n=\dfrac{(-1)^n}{\sqrt n}-\dfrac1n+O\left(\dfrac1{n^{3/2}}\right)$ et en déduire la nature de $\sum v_n$.
\end{enumerate}
\item Montrer que $\sum_{n\geq1}\dfrac{2n+1}{n^2(n+1)^2}$ converge et calculer sa somme.
\end{enumerate}

\Partie{Exercice 2 — Séries entières}

\begin{enumerate}
\item Déterminer le rayon de convergence des séries entières :
\[\text{(a)}\ \sum_{n\geq1}\ln(n)\,z^n\qquad\text{(b)}\ \sum_{n\geq1}\left(1+\frac1n\right)^{n^2}z^n\qquad\text{(c)}\ \sum_{n\geq0}\frac{z^{2n}}{2^n+n}\qquad
\text{(d)}\ \sum_{n\geq0}\frac{n!}{(2n)!}z^n.\]
\item On considère la série entière $\displaystyle\sum_{n\geq1}\frac{(-1)^{n+1}}{n(2n-1)}x^{2n}$ et on note $f$ sa fonction somme.
\begin{enumerate}[label=(\alph*)]
\item Déterminer le rayon de convergence $R$ et l'ensemble de définition de $f$.
\item Décomposer $\frac1{n(2n-1)}$ en éléments simples, puis calculer $f(x)$ pour $x\in\,]-R,R[$.
\item En déduire la valeur de $\displaystyle\sum_{n=1}^{+\infty}\frac{(-1)^{n+1}}{n(2n-1)}$.
\end{enumerate}
\item Développer en série entière au voisinage de $0$ la fonction $g:x\mapsto\ln(1+x+x^2)$ et préciser le rayon (on pourra écrire
$1+x+x^2=\frac{1-x^3}{1-x}$). En déduire $g^{(9)}(0)$.
\item Calculer $\displaystyle\sum_{n=1}^{+\infty}\frac{n^2}{2^n}$.
\end{enumerate}

\Partie{Exercice 3 — Une équation différentielle}

On considère l'équation différentielle $(E)$ : $xy''+2y'+xy=0$.
\begin{enumerate}
\item On cherche les solutions de $(E)$ développables en série entière : $y(x)=\sum_{n\geq0}a_nx^n$, de rayon $R>0$.
\begin{enumerate}[label=(\alph*)]
\item Montrer que $a_1=0$ et que $(n+1)(n+2)a_{n+1}+a_{n-1}=0$ pour tout $n\geq1$.
\item En déduire $a_n$ en fonction de $n$ et de $a_0$, le rayon de convergence, puis une expression simple de $y$.
\end{enumerate}
\item Sur $I=\,]0,+\infty[$, on pose $z(x)=xy(x)$. Montrer que $y$ est solution de $(E)$ sur $I$ si et seulement si $z''+z=0$.
En déduire toutes les solutions de $(E)$ sur $]0,+\infty[$, puis sur $]-\infty,0[$.
\item Déterminer les solutions de $(E)$ de classe $\mathcal C^2$ sur $\R$. Forment-elles un espace vectoriel de dimension $2$ ?
Pourquoi le théorème de Cauchy ne s'applique-t-il pas sur $\R$ ?
\end{enumerate}
